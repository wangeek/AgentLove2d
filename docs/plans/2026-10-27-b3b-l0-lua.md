**Step 2: fr.lua（行文本编码器，对应 fr_codec_gen.render_line 的 Lua 版）**

```lua
-- l0/fr.lua —— FR 行文本编码器（canonical .fr，spec §6；Lua 5.1/LuaJIT）
local fr = {}

function fr.esc(s)
    s = s:gsub("\\", "\\\\")
    s = s:gsub('"', '\\"')
    s = s:gsub("\n", "\\n")
    s = s:gsub("\r", "\\r")
    s = s:gsub("\t", "\\t")
    return s
end

function fr.f(v)
    if v ~= v or v == math.huge or v == -math.huge then
        error("fr: NaN/Inf 禁入数值域")
    end
    return string.format("%.6f", v)
end

function fr.tok_variant(v)
    if type(v) == "boolean" then return v and "bool 1" or "bool 0" end
    if type(v) == "number" then return "num " .. fr.f(v) end
    if type(v) == "table" then return 'str "' .. fr.esc(v[1]) .. '"' end
    error("fr: 非法 variant: " .. tostring(v))
end

-- OPS 元数据由 gen_python 生成（l0/fr_ops.lua，表驱动单一真源）：
-- fr.OPS = { dt = { fields = { {"val","f32"} } }, draw = { fields = { {"tex","u32"}, ... } }, ... }
local ops = require("fr_ops")

local put = {
    u8    = function(t, v) t[#t+1] = tostring(v) end,
    u32   = function(t, v) t[#t+1] = tostring(v) end,
    f32   = function(t, v) t[#t+1] = fr.f(v) end,
    f64   = function(t, v) t[#t+1] = fr.f(v) end,
    rgba8 = function(t, v) t[#t+1] = ("%d %d %d %d"):format(v[1], v[2], v[3], v[4]) end,
    str   = function(t, v) t[#t+1] = '"' .. fr.esc(v) .. '"' end,
    ["f32[]"] = function(t, v)
        t[#t+1] = tostring(#v)
        for _, p in ipairs(v) do t[#t+1] = fr.f(p[1]); t[#t+1] = fr.f(p[2]) end
    end,
}

function fr.render_line(op, fields)
    local spec = ops[op] or error("fr: 未知 op: " .. tostring(op))
    local t = { op }
    for _, f in ipairs(spec.fields) do
        local name, typ = f[1], f[2]
        if typ == "variant" then
            t[#t+1] = fr.tok_variant(fields[name])
        else
            put[typ](t, fields[name])
        end
    end
    if spec.variadic then
        local arr = fields[spec.variadic]
        t[#t+1] = tostring(#arr)
        for _, v in ipairs(arr) do t[#t+1] = fr.tok_variant(v) end
    end
    return table.concat(t, " ")
end

function fr.selftest()
    local samples = require("fr_samples")   -- 24 条样例（Task 1 Step 3 生成）
    for _, s in ipairs(samples) do
        print(fr.render_line(s.op, s.fields))
    end
end

return fr
```

```lua
-- l0/kernel.lua —— 极简 demo 内核
local kernel = { state = { tick = 0, hero = { x = 10.0, y = 20.0, hp = 87.0 }, banner = "L0" } }
function kernel.update(dt)
    local s = kernel.state
    s.tick = s.tick + 1
    s.hero.x = s.hero.x + 60.0 * dt
    if s.hero.x > 600 then s.hero.x = 600.0 end
end
function kernel.draw(g)
    g.clear(0, 0, 0, 255)
    g.draw(1, 0, kernel.state.hero.x, kernel.state.hero.y, 0, 1, 1, 0, 0, 255, 255, 255, 255)
    g.print(2, kernel.state.banner, 12.0, 340.0, 0, 1, 1, 255, 255, 0, 255)
    g.rectangle(0, 0.0, 0.0, 640.0, 480.0, 0.0, 0.0, 1.0, 255, 0, 0, 255)
end
return kernel
```

**run loop（排干快照，per design §4）**：帧首 `love.event.pump(); for n,a,b,c in love.event.poll() do snapshot[#snapshot+1]={n,a,b,c} end`（headless 下 OS 事件为空——**合成事件**由 runner 注入：`--inject 2:keypressed` 之类参数或直接固定脚本：帧 2 注入 keypressed，帧 3 注入 quit）→ 逐条：记录 ev 行（slot=快照序号，src=2 自定义）→ `handlers[name]` 直调（handler 内再直调另一 handler → CALL 行记深度，demo 演示一层）。quit 事件终止循环。
- dt 行每帧记录（固定值）。
- **walker.lua（状态走查器，样机口径）**：排序键深度优先遍历 `kernel.state`，值域 number/string/boolean；table 递归（路径拼接 `[i]` 或 `.key`）；产出 st 行 `st 0 <点路径> <variant>`（**obj=0 哨兵=语义样机形态，B6 A 方案换物理形态**）；s_crc = CRC32(st 行文本拼接 "\n")，d_crc = CRC32(out 行文本拼接 "\n")（**样机口径：B6 换堆遍历/真正状态快照**——两处口径写进 walker.lua docstring + Task 4 回写 spec）。
- CRC32：LuaJIT `bit` 库实现（标准多项式 0xEDB88320，查表法）。
- 会话组装（文本）：头行 `frb1 640 480 <sha64hex> base <seed>`（sha 从 Python 传入——`--sha <hex>` 参数，测试侧算好传入；demo 默认值写死当前表指纹的常量？**不写死**：main 缺省报错要求 --sha，测试传入）+ cmnt 固定文本 + frame 块（帧内行序：dt → asset(仅首帧) → in 段行(ev/call/timer) → log 段行(st/err) → out 段行(clear/draw/…) → end 段行(s_crc/d_crc)）。
  > 拍平行序说明：B0 文本形态所有行拍平进 frame 块；段归属由 op→section 元数据承载（l0/fr_ops.lua 顺带携带 section 字段——gen_python 生成时加入）。

**测试**（追加）：`--demo --frames 4 --fixed-dt 0.016667 --sha <真指纹> --out <tmp>/s.fr` → 断言：五段齐全（dt/asset 在、ev 行 src=2、call 行 depth=1、st 行 obj=0 点路径含 `hero.x`、s_crc/d_crc 非 0）；**fr-check 通过**（subprocess fr_check，含 CHECK 判据：`CHECK: frame 0`、`CHECK: s_crc ` 等）；fr-as→fr_dis→fr_as round-trip 字节一致；第二次运行产物与第一次**逐字节一致**（确定性——固定 dt+固定 cmnt+固定注入脚本）。
