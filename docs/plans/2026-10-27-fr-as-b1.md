## 会话封装 v1（本计划封版，Task 6 回写 spec §6.3）

**二进制 .frb**（小端）：

```
header（50B）:
  magic   4B   "FRB1"
  version u8   1
  width   u16
  height  u16
  sha     32B  cmd.tbl 规范表 sha256 原始摘要（与行内 TBL_SHA256 十六进制同源）
  loglvl  u8   0=off 1=base 2=full
  seed    u64  RNG 初值
entries（重复至 EOF，帧号由顺序隐含、从 0 计）:
  tag u8:
    0x01 frame → u32 body_len + body（rows 逐条，各以 opcode 起始；0xFF 禁入帧内）
    0xFF cmnt  → str（u32 字节数 + UTF-8 原文；禁 \r \n）
    其它 tag → SessionError
```

**文本 .fr**（canonical：LF、无行尾空白、UTF-8）：

```
frb1 <w> <h> <sha64hex小写> <off|base|full> <seed十进制>
cmnt <原文（"cmnt " 之后逐字保留，禁 \r \n）>          ← 0..n 条，可在任意两帧之间
frame 0                                               ← n 必须连续（0,1,2…），否则报错
<row 行（fr_codec_gen.render_line 输出；asset 用规范形）>
frame 1
...
```

**asset 行文本规范形**（kind 用名字、二进制用数字——双形态各取所长；hex 为小写连续十六进制，空数据记 `-`）：

```
asset image  <id> <w> <h> <hex>
asset canvas <id> <w> <h>
asset quad   <id> <x%.6f> <y%.6f> <w%.6f> <h%.6f> <sw%.6f> <sh%.6f>
asset font   <id> <size%.6f> <hex>
```

**错误口径**：文本侧 SessionError 含行号；二进制侧含偏移；行级错误透传 row codec 的 ValueError（已在 B0 封版）。

```python
# tools/fr_session.py
"""会话级封装（B1 封版）：.fr 文本 ↔ Session ↔ .frb 二进制。

Session = dict(width, height, sha, loglvl, seed,
               entries=[("cmnt", text) | ("frame", [(op, fields), ...]), ...])
布局与错误口径见 docs/fr-spec.md §6.3。行级编解码来自 fr_codec_gen（表生成）；
本模块是结构性行（CMNT/帧定界），手写——不随 cmd.tbl 变化。
"""
import binascii, re
from tools import fr_codec_gen as _rc

MAGIC = b"FRB1"
VERSION = 1
_HEADER_LEN = 4 + 1 + 2 + 2 + 32 + 1 + 8
TAG_FRAME = 0x01
TAG_CMNT = 0xFF
_LOGLVL = {"off": 0, "base": 1, "full": 2}
_LOGLVL_NAME = {v: k for k, v in _LOGLVL.items()}
_HEX_EMPTY = "-"

class SessionError(Exception):
    pass

# ---------- 文本 ----------

def parse_text(text):
    lines = text.split("\n")
    if lines and lines[-1] == "":
        lines.pop()
    else:
        raise SessionError("文本必须以单个 LF 结尾")
    hdr = lines[0].split(" ")
    if hdr[0] != "frb1" or len(hdr) != 6:
        raise SessionError("第 1 行：会话头必须为 'frb1 w h sha loglvl seed'")
    w, h, sha, loglvl, seed = hdr[1], hdr[2], hdr[3], hdr[4], hdr[5]
    if not (w.isdigit() and h.isdigit() and 0 < int(w) <= 65535 and 0 < int(h) <= 65535):
        raise SessionError("第 1 行：几何非法")
    if not re.fullmatch(r"[0-9a-f]{64}", sha):
        raise SessionError("第 1 行：sha 必须为 64 位小写十六进制")
    if loglvl not in _LOGLVL:
        raise SessionError("第 1 行：loglvl 非法 %r" % loglvl)
    if not seed.isdigit() or int(seed) >= 1 << 64:
        raise SessionError("第 1 行：seed 非法")
    s = dict(width=int(w), height=int(h), sha=sha, loglvl=loglvl, seed=int(seed), entries=[])
    cur = None  # 当前帧 rows
    for no, line in enumerate(lines[1:], start=2):
        if line == "":
            raise SessionError("第 %d 行：空行禁入 canonical" % no)
        if line.startswith("cmnt "):
            body = line[5:]
            if "\r" in body:
                raise SessionError("第 %d 行：cmnt 禁含 \\r" % no)
            if cur is not None:
                raise SessionError("第 %d 行：cmnt 禁入帧内" % no)
            s["entries"].append(("cmnt", body))
        elif line.startswith("frame "):
            n = line[6:]
            idx = sum(1 for k, _ in s["entries"] if k == "frame")
            if not n.isdigit() or int(n) != idx:
                raise SessionError("第 %d 行：帧号必须为 %d（连续）" % (no, idx))
            cur = []
            s["entries"].append(("frame", cur))
        else:
            if cur is None:
                raise SessionError("第 %d 行：行必须位于 frame 块内" % no)
            try:
                cur.append(_rc.parse_line(line))
            except ValueError as e:
                raise SessionError("第 %d 行：%s" % (no, e)) from e
    return s

def dump_text(s):
    out = ["frb1 %d %d %s %s %d" % (s["width"], s["height"], s["sha"],
                                    _LOGLVL[s["loglvl"]], s["seed"])]
    idx = 0
    for kind, payload in s["entries"]:
        if kind == "cmnt":
            if "\r" in payload or "\n" in payload:
                raise SessionError("cmnt 禁含 \\r \\n")
            out.append("cmnt " + payload)
        else:
            out.append("frame %d" % idx); idx += 1
            for op, fields in payload:
                out.append(_rc.render_line(op, fields))
    return "\n".join(out) + "\n"

# ---------- 二进制 ----------

def parse_binary(data):
    if data[:4] != MAGIC:
        raise SessionError("off 0：magic 错误（需 FRB1）")
    if len(data) < _HEADER_LEN:
        raise SessionError("off 0：头部截断")
    if data[4] != VERSION:
        raise SessionError("off 4：版本不支持 %d" % data[4])
    off = 5
    w = int.from_bytes(data[off:off+2], "little"); off += 2
    h = int.from_bytes(data[off:off+2], "little"); off += 2
    sha = data[off:off+32].hex(); off += 32
    lvl = data[off]; off += 1
    if lvl not in _LOGLVL_NAME:
        raise SessionError("off %d：loglvl 非法 %d" % (off - 1, lvl))
    seed = int.from_bytes(data[off:off+8], "little"); off += 8
    s = dict(width=w, height=h, sha=sha, loglvl=_LOGLVL_NAME[lvl], seed=seed, entries=[])
    while off < len(data):
        tag = data[off]; off += 1
        if tag == TAG_CMNT:
            n = int.from_bytes(data[off:off+4], "little"); off += 4
            body = data[off:off+n].decode("utf-8"); off += n
            s["entries"].append(("cmnt", body))
        elif tag == TAG_FRAME:
            n = int.from_bytes(data[off:off+4], "little"); off += 4
            body = data[off:off+n]; off += n
            rows, mv, i = [], memoryview(body), 0
            while i < len(body):
                try:
                    op, fields = _rc.decode_row(mv[i:])
                except ValueError as e:
                    raise SessionError("帧内 off %d：%s" % (i, e)) from e
                rows.append((op, fields))
                # decode_row 消耗整行：求行长——用重编码长度（B0 无 cursor 变体，见下注）
                i += len(_rc.encode_row(op, fields))
            s["entries"].append(("frame", rows))
        else:
            raise SessionError("off %d：entry tag 非法 0x%02X" % (off - 1, tag))
    return s

def dump_binary(s):
    out = [MAGIC, bytes([VERSION]),
           s["width"].to_bytes(2, "little"), s["height"].to_bytes(2, "little"),
           binascii.unhexlify(s["sha"]),
           bytes([_LOGLVL[s["loglvl"]]]), s["seed"].to_bytes(8, "little")]
    for kind, payload in s["entries"]:
        if kind == "cmnt":
            b = payload.encode("utf-8")
            out.append(bytes([TAG_CMNT]) + len(b).to_bytes(4, "little") + b)
        else:
            body = b"".join(_rc.encode_row(op, f) for op, f in payload)
            out.append(bytes([TAG_FRAME]) + len(body).to_bytes(4, "little") + body)
    return b"".join(out)
```

实现注记（执行者照做）：①帧内行消耗长度用"重编码长度"是 B0 无 cursor 变体的务实解，正确性由 round-trip 测试保证；代价是 decode 两次解一行——B1 规模下无关紧要，fr-spec §6.3 注明"B1 之后若需流式可给 decode_row 加 cursor 变体"（终审建议②）；②`test_session` 的 tag 非法用例写 offset 50——与 `_HEADER_LEN=50` 对应。

```python
# tools/fr_as.py
#!/usr/bin/env python3
"""fr-as：.fr 文本 → .frb 二进制。用法: fr_as <in.fr> [-o out.frb]（缺省 stdout 不适用二进制，-o 必填）。"""
import os, sys
sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))
from tools import fr_session as fs

def main(argv):
    args = [a for a in argv]
    out = None
    if "-o" in args:
        i = args.index("-o"); out = args[i + 1]; del args[i:i + 2]
    if len(args) != 1 or out is None:
        print(__doc__.strip(), file=sys.stderr); return 2
    try:
        s = fs.parse_text(open(args[0], encoding="utf-8").read())
        open(out, "wb").write(fs.dump_binary(s))
    except (fs.SessionError, ValueError, OSError) as e:
        print("fr-as: %s" % e, file=sys.stderr); return 1
    return 0

if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))
```
