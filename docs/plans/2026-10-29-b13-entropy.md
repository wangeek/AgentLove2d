```lua
local State = require("src.core.state")
local C = require("src.constants")
State.switch(C.MODE.BATTLE)

local t = 0
local orig_update = love.update
function love.update(dt)
    if orig_update then orig_update(dt) end
    t = t + dt
    if t >= 0.5 then
        t = 0
        -- A 路：点击按钮中心（坐标来自 Step 3 探查；下例为占位，须替换实测值）
        -- love.event.push("mousepressed", BTN_X, BTN_Y, 1, false, 1)
        -- B 路：直调推进（防御式）
        local h = State.current and State.current() or nil
        if h and h.advanceTurn then pcall(function() h:advanceTurn() end) end
    end
end
io.stderr:write("fr-boot-battle: switched + advance loop\n")
```

```python
# tools/fr_stats.py
"""fr-stats —— 流熵剖面（B13 字节经济度量基建）。

用法：python tools/fr_stats.py <in.fr|in.frb> [--json]

四层剖面（设计真源 docs/plans/2026-10-29-b13-entropy-roadmap-design.md §2.2）：
  op 层   每 op 行数/字节占比/均行长——压缩火力对准谁
  参数层 默认值命中率（B14 省略表准入证据，门槛 >90%）
  量化层 模拟 px/q64 编码字节（含省略叠加、gzip 参照）
  时间层 帧间整行重复率 / 行集 diff 分布（B15 空间）
"""
import json
import sys

import tools.fr_session as fs


def profile(path):
    text = open(path, "rb").read().decode("utf-8")
    sess = fs.parse_text(text)
    ops = {}
    frames_out = []  # 每帧行集（时间层用）
    out_bytes_total = 0
    for kind, rows in sess["entries"]:
        if kind != "frame":
            continue
        frows = [r for r in rows if r[0] in (
            "clear", "draw", "print", "rect", "line", "circle",
            "poly", "arc", "points", "rt", "scissor")]
        frames_out.append(frows)
        for op, f in frows:
            s = _render(op, f)
            e = ops.setdefault(op, {"rows": 0, "bytes": 0})
            e["rows"] += 1
            e["bytes"] += len(s.encode()) + 1
            out_bytes_total += len(s.encode()) + 1
    for v in ops.values():
        v["avg"] = round(v["bytes"] / v["rows"], 1)
    return {"ops": ops, "out_bytes_total": out_bytes_total,
            "_frames": frames_out}
```
（`_render` 用 fr_codec_gen.render_line；defaults/量化/时间层在后续步补——先让 op 层测试绿。）

```python
def time_layer(profile_result):
    frames = profile_result["_frames"]
    same = sum(1 for a, b in zip(frames, frames[1:]) if a == b)
    diffs = [len(set(b) - set(a)) for a, b in zip(frames, frames[1:])]
    return {"frames": len(frames), "same_ratio": same / max(1, len(frames) - 1),
            "avg_diff_rows": sum(diffs) / max(1, len(diffs))}
```
