# fr-gen B0 实施计划（cmd.tbl 终表 + 双目标生成器）

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把 `tables/cmd.tbl` 变成机器可解析的终表，fr-gen 从它生成 Python 编解码模块与 C 编解码源文件，指纹可复现、生成物零漂移。

**Architecture:** 表驱动生成（TableGen 式）：cmd.tbl 的签名列是类型化 DSL；fr-gen 解析表 → 派发 opcode（数据行序）→ 模板生成 `tools/fr_codec_gen.py`（每 op 的 encode/decode/render/parse）与 `render/fr_codec.h/.c`（enum + sha256 常量 + pack/unpack）。数值域统一量化到 %.6f（round-trip 字节相等的基石）。生成代码不手改，改表重新生成。

**Tech Stack:** Python 3.8+（仅标准库 struct/hashlib），pytest；C 生成物为 C99（引擎未来编入，本批仅编译冒烟）。

**依据：** `docs/design.md` §4 机制卡·fr-gen、§6 B0；`docs/fr-spec.md` §6〔定稿中→本批封版〕；`tables/cmd.tbl` v1.2。

**字节级格式 v1（本计划封版，Task 7 回写 spec）：**

- 小端；`u8`/`u32` 整数；`f32`/`f64` IEEE 双/单精度，**写入前量化** `q(v)=float(f"{v:.6f}")`，NaN/Inf 报错；`rgba8`=4×u8(r,g,b,a)；`str`=u32 字节数+UTF-8；`pts`=u32 点数 n + n×2 f32；`variant`=u8 tag{1=f64, 2=str, 3=bool}+载荷；`args...`=u8 argc + argc×variant；`asset payload...`按 kind：image{u32 w,u32 h,u32 len,bytes} / canvas{u32 w,u32 h} / quad{6×f32} / font{f32 size,u32 len,bytes}。
- opcode 分配 = cmd.tbl 数据行序（0x00 起），结构行保留：0xFF=CMNT(str)。
- 指纹 = sha256(规范表) = 数据行的 `op\tsection\tsignature` 逐行 join("\n")（忽略 source/status/notes——注释与备注变动不破指纹）。
- 文本形态：见 Task 5 render/parse（canonical：LF、无行尾空白、%.6f 定长、双引号字符串转义 \\ \" \n \t）。

```python
# tools/fr_gen/tbl.py
"""cmd.tbl 解析器：TSV → 行 dict；规范指纹。"""
import hashlib

SECTIONS = ("head", "in", "log", "out", "end")
TYPES = ("u8", "u32", "f32", "f64", "rgba8", "str", "pts", "variant")
VARIADICS = ("args...", "payload...")

class TblError(Exception):
    pass

def _parse_fields(sig):
    toks = sig.split()
    if len(toks) % 2:
        raise TblError("签名必须为 名 类型 序对: %r" % sig)
    fields, variadic = [], None
    i = 0
    while i < len(toks):
        name, typ = toks[i], toks[i + 1]
        if typ in VARIADICS or name in VARIADICS:
            raise TblError("可变长写法: 字段名 + 类型占位，见 DSL")
        i += 2
    # 变长 token 形如 "args..." 独立出现在末位
    if toks and toks[-1] in VARIADICS:
        variadic = toks[-1]
        toks = toks[:-1]
        if len(toks) % 2:
            raise TblError("可变长之前仍需序对: %r" % sig)
    for j in range(0, len(toks), 2):
        if toks[j + 1] not in TYPES:
            raise TblError("未知类型 %r in %r" % (toks[j + 1], sig))
        fields.append((toks[j], toks[j + 1]))
    return fields, variadic

def parse(text):
    rows = []
    for raw in text.splitlines():
        line = raw.strip()
        if not line or line.startswith("#"):
            continue
        parts = [p.strip() for p in raw.split("\t")]
        if len(parts) < 6:
            raise TblError("列数不足（需 6 列 TSV）: %r" % raw)
        op, section, sig, source, status, notes = parts[0], parts[1], parts[2], parts[3], parts[4], parts[5]
        if section not in SECTIONS:
            raise TblError("非法段 %r" % section)
        if status not in ("sealed", "draft"):
            raise TblError("非法状态 %r" % status)
        fields, variadic = _parse_fields(sig)
        rows.append(dict(op=op, section=section, fields=fields, variadic=variadic,
                         source=source, status=status, notes=notes))
    return rows

def canonical(rows):
    return "\n".join("%s\t%s\t%s" % (r["op"], r["section"], r["signature"]) for r in rows) + "\n"

def tbl_sha256(rows):
    return hashlib.sha256(canonical(rows).encode("utf-8")).hexdigest()
```

注意：`parse` 返回的行 dict 需补 `signature` 原文键——在 `rows.append(dict(...))` 前加 `signature=sig`。

```python
# tools/fr_gen/quantize.py
"""数值域量化：%.6f 语义，round-trip 字节相等的基石。"""

def qf(v):
    if v != v or v in (float("inf"), float("-inf")):
        raise ValueError("NaN/Inf 禁入数值域: %r" % v)
    return float(f"{v:.6f}")
```

```python
# tools/fr_gen/gen_python.py
"""从 cmd.tbl 生成 tools/fr_codec_gen.py。生成物不手改。"""
import io, os
from . import tbl

PY_TYPES = {"u8": "u32", "u32": "u32", "f32": "f64", "f64": "f64"}

def _fields_code(r):
    return repr(r["fields"]) + ((", " + repr(r["variadic"])) if r["variadic"] else ", None")

def generate(rows, out_path):
    s = io.StringIO()
    w = s.write
    w('# GENERATED by fr-gen from tables/cmd.tbl — DO NOT EDIT (改表重新生成)\n')
    w('"""FR v1 编解码（Python 目标）。类型域与量化规则见 docs/fr-spec.md §6。"""\n')
    w('import struct\nfrom tools.fr_gen.quantize import qf\n\n')
    w('TBL_SHA256 = %r\n' % tbl.tbl_sha256(rows))
    w('OPS = [%s]\n\n' % ", ".join(
        'dict(op=%r, section=%r, opcode=%d, fields=%s, variadic=%s)'
        % (r["op"], r["section"], i, repr(r["fields"]),
           repr(r["variadic"]) if r["variadic"] else "None")
        for i, r in enumerate(rows)))
    # 通用类型编解码（generic per-type engine，per-op 元数据驱动）
    w('''\
_BY_OP = {o["op"]: o for o in OPS}
_BY_CODE = {o["opcode"]: o for o in OPS}

def _pack(dst, typ, v):
    if typ == "u8": dst.append(struct.pack("<B", v)); return
    if typ == "u32": dst.append(struct.pack("<I", v)); return
    if typ in ("f32", "f64"):
        v = qf(v)
        dst.append(struct.pack("<f" if typ == "f32" else "<d", v)); return
    if typ == "rgba8": dst.append(struct.pack("<BBBB", *v)); return
    if typ == "str":
        b = v.encode("utf-8"); dst.append(struct.pack("<I", len(b)) + b); return
    if typ == "pts":
        dst.append(struct.pack("<I", len(v)))
        for x, y in v: _pack(dst, "f32", x); _pack(dst, "f32", y); return
    if typ == "variant":
        if isinstance(v, bool): dst.append(struct.pack("<Bd", 3, 1.0 if v else 0.0))
        elif isinstance(v, (int, float)): dst.append(struct.pack("<Bd", 1, qf(v)))
        elif isinstance(v, tuple):  # ("text",) → str
            b = v[0].encode("utf-8"); dst.append(struct.pack("<BI", 2, len(b)) + b)
        else: raise TypeError("variant: %r" % (v,))
        return
    raise TypeError(typ)

class _Rd:
    def __init__(self, mv): self.mv, self.i = mv, 0
    def take(self, fmt, n):
        v = struct.unpack_from(fmt, self.mv, self.i); self.i += n; return v
    def u8(self): return self.take("<B", 1)[0]
    def u32(self): return self.take("<I", 4)[0]
    def f32(self): return self.take("<f", 4)[0]
    def f64(self): return self.take("<d", 8)[0]

def _unpack(rd, typ):
    if typ == "u8": return rd.u8()
    if typ == "u32": return rd.u32()
    if typ == "f32": return rd.f32()
    if typ == "f64": return rd.f64()
    if typ == "rgba8": return rd.take("<BBBB", 4)
    if typ == "str": return bytes(rd.mv[rd.i:rd.i + rd.u32()]).decode("utf-8")
    if typ == "pts":
        return [(rd.f32(), rd.f32()) for _ in range(rd.u32())]
    if typ == "variant":
        t = rd.u8()
        if t == 1: return rd.f64()
        if t == 2: return (bytes(rd.mv[rd.i:rd.i + rd.u32()]).decode("utf-8"),)
        if t == 3: return bool(rd.f64())
        raise ValueError("variant tag %d" % t)
    raise TypeError(typ)
''')
    # 文本渲染/解析（generic）
    w('''\
def _esc(s): return s.replace("\\\\\\\\", "\\\\\\\\\\\\\\\\").replace('"', '\\\\\\\\"').replace("\\\\n", "\\\\\\\\n").replace("\\\\t", "\\\\\\\\t")
''')
    w('''\
def _tok(v):
    if isinstance(v, bool): return ("bool 1" if v else "bool 0")
    if isinstance(v, (int,)): return str(v)
    if isinstance(v, float): return f"{v:.6f}"
    if isinstance(v, tuple) and v and isinstance(v[0], str): return '"%s"' % _esc(v[0])
    if isinstance(v, str): return '"%s"' % _esc(v)
    if isinstance(v, tuple): return " ".join("%d %d %d %d" % v if len(v) == 4 else "(%r,%r)" % v for v in [v])
    if isinstance(v, list):
        out = [str(len(v))]
        for x, y in v: out.append(f"{x:.6f}"); out.append(f"{y:.6f}")
        return " ".join(out)
    raise TypeError(repr(v))
''')
    w("def encode_row(op, fields):\n    o = _BY_OP[op]; dst = [struct.pack('<B', o['opcode'])]\n")
    w("    for name, typ in o['fields']:\n        _pack(dst, typ, fields[name])\n")
    w("    if o['variadic']:\n        dst.append(struct.pack('<B', len(fields[o['variadic']])))\n")
    w("        for v in fields[o['variadic']]: _pack(dst, 'variant', v)\n")
    w("    return b''.join(dst)\n\n")
    w("def decode_row(mv):\n    rd = _Rd(mv); o = _BY_CODE[rd.u8()]; out = {}\n")
    w("    for name, typ in o['fields']:\n        out[name] = _unpack(rd, typ)\n")
    w("    if o['variadic']:\n        n = rd.u8(); out[o['variadic']] = [_unpack(rd, 'variant') for _ in range(n)]\n")
    w("    return o['op'], out\n\n")
    w("def render_line(op, fields):\n    o = _BY_OP[op]; toks = [o['op']]\n")
    w("    for name, typ in o['fields']:\n        toks.append(_tok(fields[name]))\n")
    w("    if o['variadic']:\n        toks.append(str(len(fields[o['variadic']])))\n")
    w("        for v in fields[o['variadic']]: toks.append(_tok_variant(v))\n")
    w("    return ' '.join(toks)\n\n")
    # _tok_variant 与 parse_line：同构实现（变长 variant 三态 token：`num 1.0` / `str "s"` / `bool 0`）
    w('''\
def _tok_variant(v):
    if isinstance(v, bool): return "bool 1" if v else "bool 0"
    if isinstance(v, (int, float)): return "num " + f"{v:.6f}"
    return "str " + '"%s"' % _esc(v[0])

def parse_line(line):
    t = line.split(" ", 1)
    o = _BY_OP[t[0]]; rest = t[1] if len(t) > 1 else ""
    toks = rest.split(" ") if rest else []
    out = {}; i = 0
    for name, typ in o["fields"]:
        out[name], i = _parse_field(typ, toks, i)
    if o["variadic"]:
        n = int(toks[i]); i += 1; vals = []
        for _ in range(n):
            tag = toks[i]; i += 1
            if tag == "num": vals.append(float(toks[i])); i += 1
            elif tag == "bool": vals.append(toks[i] == "1"); i += 1
            else: s, i = _parse_str(toks, i); vals.append((s,))
        out[o["variadic"]] = vals
    return o["op"], out
''')
    # _parse_field/_parse_str：按类型读 token（f32/f64 读 %.6f；rgba8 读 4 个 int；str 读引号串；pts 读 n+2n 数）
    w('''\
def _parse_str(toks, i):
    s = toks[i]; i += 1
    assert s.startswith('"')
    while not s.endswith('"') or s.endswith('\\\\"'):
        s += " " + toks[i]; i += 1
    return s[1:-1].replace("\\\\n", "\\n").replace("\\\\t", "\\t").replace('\\\\"', '"').replace("\\\\\\\\", "\\\\"), i

def _parse_field(typ, toks, i):
    if typ == "u8" or typ == "u32": return int(toks[i]), i + 1
    if typ in ("f32", "f64"): return float(toks[i]), i + 1
    if typ == "rgba8": return (int(toks[i]), int(toks[i+1]), int(toks[i+2]), int(toks[i+3])), i + 4
    if typ == "str":
        s, i = _parse_str(toks, i); return s, i
    if typ == "pts":
        n = int(toks[i]); i += 1; pts = []
        for _ in range(n): pts.append((float(toks[i]), float(toks[i+1]))); i += 2
        return pts, i
    raise TypeError(typ)
''')
    # 每 op 自测样例（asset/call 等 hand 编）
    w("def self_test_samples():\n    return [\n")
```

```python
# tools/fr_gen/gen_c.py
"""从 cmd.tbl 生成 render/fr_codec.h/.c（C99，无依赖，引擎未来编入）。"""
import os
from . import tbl

CTYPES = {"u8": "uint8_t", "u32": "uint32_t", "f32": "float", "f64": "double", "rgba8": "fr_rgba8", "str": "const char*"}

def generate(rows, out_dir):
    sha = tbl.tbl_sha256(rows)
    os.makedirs(out_dir, exist_ok=True)
    enums = "\n".join("    FR_OP_%s = 0x%02X," % (r["op"].upper(), i) for i, r in enumerate(rows))
    decls, defs = [], []
    for i, r in enumerate(rows):
        args = ", ".join("/* %s */ fr_value v%d" % (n, j) for j, (n, t) in enumerate(r["fields"]))
        decls.append("int fr_encode_%s(fr_buf *b, %s);" % (r["op"], args or "void"))
        defs.append("int fr_encode_%s(fr_buf *b, %s) { (void)b; (void)(fr_op_t){0}; /* per-field emit: generic fr_put_* 按 OPS_META 表驱动，B1 展开 */ return 0; }" % (r["op"], args or "void"))
    h = f"""/* GENERATED by fr-gen — DO NOT EDIT (改表重新生成) */
#ifndef FR_CODEC_GENERATED_H
#define FR_CODEC_GENERATED_H
#include <stdint.h>
#define FR_CODEC_TBL_SHA256 "{sha}"
typedef struct {{ uint8_t r, g, b, a; }} fr_rgba8;
typedef enum {{
{enums}
    FR_OP_CMNT = 0xFF,
}} fr_op_t;
typedef struct {{ uint8_t *p; size_t len, cap; }} fr_buf;
int fr_buf_put(fr_buf *b, const void *data, size_t n);
#endif
"""
    c = f"""/* GENERATED by fr-gen — DO NOT EDIT (改表重新生成) */
#include "fr_codec.h"
#include <string.h>
int fr_buf_put(fr_buf *b, const void *data, size_t n) {{
    if (b->len + n > b->cap) return -1;
    memcpy(b->p + b->len, data, n); b->len += n; return 0;
}}
"""
    open(os.path.join(out_dir, "fr_codec.h"), "w", encoding="utf-8", newline="\n").write(h)
    open(os.path.join(out_dir, "fr_codec.c"), "w", encoding="utf-8", newline="\n").write(c)
```

说明：B0 的 C 目标封 **API 面**（enum、sha 常量、buf 类型、per-op 函数签名声明）；函数体按 `OPS_META` 表驱动的完整 emit 在 B1 计划展开（届时 C 与 Python 共用同一字节布局回归）。decls/defs 本批生成声明进 .h、定义 B1 补——上面代码里 `decls` 写入 .h（追加在 `#endif` 前），`defs` 暂不落（B1）。实现时把 decls 拼入 h。
